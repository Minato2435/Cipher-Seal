### Task 2: Firestore security rules

**Files:**
- Modify: `firestore.rules` (replace the deny-all placeholder)
- Create: `docs/part3-rules-notes.md`

**Interfaces:**
- Consumes: the collection shapes from Part 2 (`users`, `publicKeys`, `sessions`, `messages`, `riskScores`, `securityEvents`, `policyActions`, `alerts`).
- Produces: participant-/owner-scoped client reads; all writes remain functions-only (Admin SDK bypasses rules).

- [ ] **Step 1: Write `firestore.rules`**

```
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function isAdmin()  { return signedIn() && request.auth.token.role == 'admin'; }
    function me()       { return request.auth.uid; }

    // Own profile; admins see everyone. No client writes (functions set role/status).
    match /users/{uid} {
      allow get: if me() == uid || isAdmin();
      allow list: if isAdmin();
      allow write: if false;
    }

    // Any signed-in user may read public keys (needed to start a session). No writes.
    match /publicKeys/{uid} {
      allow read: if signedIn();
      allow write: if false;
    }

    // Wrapped private keys: never reachable from a client.
    match /privateKeys/{uid} {
      allow read, write: if false;
    }

    // Sessions you take part in.
    match /sessions/{sessionId} {
      allow read: if signedIn() && (me() in resource.data.participants || isAdmin());
      allow write: if false;
    }

    // Messages you sent or received. The client query MUST filter
    // `where('participants', 'array-contains', uid)` for this to be allowed.
    match /messages/{messageId} {
      allow read: if signedIn() && (me() in resource.data.participants || isAdmin());
      allow write: if false;
    }

    // Your own risk score; admins see all.
    match /riskScores/{uid} {
      allow get: if me() == uid || isAdmin();
      allow list: if isAdmin();
      allow write: if false;
    }

    // Monitoring collections: admin read only.
    match /securityEvents/{id} { allow read: if isAdmin(); allow write: if false; }
    match /policyActions/{id}  { allow read: if isAdmin(); allow write: if false; }
    match /alerts/{id}         { allow read: if isAdmin(); allow write: if false; }

    // Anything else: closed.
    match /{document=**} { allow read, write: if false; }
  }
}
```

- [ ] **Step 2: Write `docs/part3-rules-notes.md`**

Document: rules apply only to the client SDK (the Admin SDK in Cloud Functions bypasses them, so tests are unaffected); every client query must carry a filter the rule can verify (`messages` → `array-contains` on `participants`; `sessions` → `array-contains` on `participants`; `riskScores` → `doc(uid)` by id, no list unless admin); `role`/`status` custom claims come from `grant_admin.py` / `admin_set_status` and take effect on the client's next ID-token refresh (call `getIdToken(true)` after status changes).

```ts
import { httpsCallable, FunctionsError } from "firebase/functions";
import { functions } from "./firebase";

export type ApiErrorCode =
  | "REAUTH_REQUIRED" | "ACCOUNT_BLOCKED" | "SIGNATURE_INVALID"
  | "PEER_NOT_READY" | "NOT_PARTICIPANT" | "FORBIDDEN" | "SESSION_INACTIVE"
  | "INVALID_KIND" | "UNAUTHENTICATED" | "UNKNOWN";

export class ApiError extends Error {
  constructor(public code: ApiErrorCode, message: string) { super(message); }
}

// Cloud Functions prefixes HttpsError messages with "<CODE>: <text>".
function toApiError(e: unknown): ApiError {
  const fe = e as FunctionsError;
  const raw = fe?.message ?? String(e);
  const m = /^([A-Z_]+):\s*(.*)$/.exec(raw);
  if (m) return new ApiError(m[1] as ApiErrorCode, m[2]);
  if (fe?.code === "functions/unauthenticated") return new ApiError("UNAUTHENTICATED", raw);
  return new ApiError("UNKNOWN", raw);
}

async function call<T>(name: string, data?: unknown): Promise<T> {
  try {
    const res = await httpsCallable(functions, name)(data);
    return res.data as T;
  } catch (e) {
    throw toApiError(e);
  }
}

export const api = {
  registerKeys: (displayName?: string) => call<{ created: boolean }>("register_keys", { displayName }),
  establishSession: (peerUid: string) => call<{ sessionId: string }>("establish_session", { peerUid }),
  sendMessage: (sessionId: string, plaintext: string) => call<{ messageId: string }>("send_message", { sessionId, plaintext }),
  readMessage: (messageId: string) => call<{ plaintext: string }>("read_message", { messageId }),
  reauth: (password: string) => call<{ ok: boolean }>("reauth", { password }),
  simulateAttack: (kind: string, targetUid?: string) => call<{ events: number }>("simulate_attack", { kind, targetUid }),
  adminSetStatus: (uid: string, status: string) => call<{ ok: boolean }>("admin_set_status", { uid, status }),
};
```

### Task 9: Composer + transformation strip + re-auth + locked state  ⟵ signature element #2

**Files:**

- [ ] **Step 1: `TransformationStrip.tsx`** — the lit instrument. A dark (`--instrument`) panel docked under the composer, `font-mono`, `--phosphor` for signal. Four rows, each with a left label in `--ink-soft` small-caps: `plain` · `ML-KEM` · `AES-GCM` · `ML-DSA`. Props: `{ phase: 'idle'|'sending'|'done'|'error', plaintext, sessionFp, ct, sig, verified }`.
  - **idle:** all rows dim, values `—`.
  - **sending → done (staged, ~700ms total, reduced-motion = instant):**
    1. `plain` fills with the typed text (in quotes), row brightens.
    2. `ML-KEM` row: a 6-cell block of `▚▞` glyphs shuffles for ~150ms then settles to `sessionFp` (16 hex).
    3. `AES-GCM` row: `ct` streams in as grouped hex, left to right, clipped to one line with a middle-ellipsis.
    4. `ML-DSA` row: a horizontal bar wipes left→right (~200ms) then a `✓` stamps at the end with a tiny scale-pop; label shows `{sigBytes}-byte seal`.
  - **error:** the `ML-DSA` row turns `--high`, shows "seal rejected".
  Use CSS transitions + a small `useEffect` state machine keyed on `phase`. No animation library.

- [ ] **Step 2: `Composer.tsx`** — a textarea (mono, auto-grow to 4 lines) + a "send" button (IBM Plex Sans 600). On send:
  1. set strip `phase='sending'`, pass the plaintext through.
  2. `await api.sendMessage(sessionId, text)`.
  3. read back the just-written `messages/{id}` doc for `ct_b64`/`sig_b64`; compute `sessionFp` from the session's `kemCtB64` via `fingerprint()`.
  4. strip `phase='done'` with those values; clear the textarea.
  - On `ApiError`:
    - `REAUTH_REQUIRED` → open `ReAuthModal`; on success retry the send once.
    - `ACCOUNT_BLOCKED` → set a `locked` state: disable the textarea + button, show "This account is locked. An administrator can restore it." in `--critical` above the composer. (The `RiskFrame` will already be CRITICAL.)
    - `SESSION_INACTIVE` → inline "This session has ended. Start a new one to keep talking."
    - else → inline "Couldn't send. Try again." and strip `phase='error'`.

- [ ] **Step 3: `ReAuthModal.tsx`** — a small centered dialog (`role="dialog"`, focus-trapped, `Esc` closes). Copy: title "Confirm it's you", body "Your recent activity looked unusual. Re-enter your password to keep sending.", one password field, "Confirm" button → `api.reauth(password)`. On `ok` close + resolve; on failure "That password didn't match." On close without success, the send is abandoned (strip returns to idle).
